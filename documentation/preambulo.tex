% Pie de tabla/figura fuera de flotante y línea de fuente
% "Tabla 1." / "Figura 1." con punto, como en plantilla.docx (no "Tabla 1:")
\captionsetup{justification=raggedright,singlelinecheck=false,labelsep=period}
\newcommand{\fuente}[1]{{\par\raggedright\footnotesize\itshape #1\par}\medskip}

% longtable reserva un número de tabla al abrirse aunque el título se coloque
% antes con \captionof, y la numeración acababa saltando de dos en dos
% (Tabla 1, Tabla 3, Tabla 5...). Se guarda el contador al entrar y se
% restaura al salir para que la secuencia sea correlativa.
\usepackage{etoolbox}
\AtBeginEnvironment{longtable}{\xdef\tablanumguardado{\the\value{table}}}
\AtEndEnvironment{longtable}{\setcounter{table}{\tablanumguardado}}
\setlist{itemsep=2pt,topsep=4pt}
